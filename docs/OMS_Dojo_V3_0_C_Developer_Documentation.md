## Architecture Overview

### Unified Application
- Single HTML file with ALL functionality
- 5 tabs: Controls, Presets, Library, Code, LIVE
- No external dependencies
- Runs entirely client-side
- No server required

### Major Systems
1. **WebGPU Rendering** - Shader compilation and execution
2. **Audio Analysis** - Web Audio API with FFT
3. **File Management** - LIBRARY folder via File System Access API
4. **Performance Interface** - LIVE tab with grid/mixer
5. **State Management** - .oms file format + Control States

function switchTab(tabName) {
    // Hide all tabs
    document.querySelectorAll('.tab-content > div').forEach(div => {
        div.style.display = 'none';
    });
    
    if (tabName === 'live') {
        // V3.0.0-B: Special LIVE tab handling
        
        // LEFT: Show Library
        document.getElementById('libraryTabContent').style.display = 'block';
        
        // RIGHT: Hide canvas/audio, show LIVE mixer
        document.getElementById('renderCanvas').style.display = 'none';
        document.querySelector('.audio-controls-strip').style.display = 'none';
        document.getElementById('liveMixerContainer').style.display = 'flex';
        
        // Initialize LIVE tab on first view
        if (!liveTabInitialized) {
            initLiveTab();
            liveTabInitialized = true;
        }
    } else {
        // Other tabs: Show selected tab content
        document.getElementById(`${tabName}TabContent`).style.display = 'block';
        
        // RIGHT: Show canvas/audio, hide LIVE mixer
        document.getElementById('renderCanvas').style.display = 'block';
        document.querySelector('.audio-controls-strip').style.display = 'flex';
        document.getElementById('liveMixerContainer').style.display = 'none';
    }
    
    // Update active tab button
    document.querySelectorAll('.tab-bar button').forEach(btn => {
        btn.classList.remove('active');
    });
    event.target.classList.add('active');
}

### Rotary Knob Rendering (V3.0.0-A)

```javascript
function drawRotaryKnob(canvas, value) {
    const ctx = canvas.getContext('2d');
    const centerX = 30, centerY = 30, radius = 25;
    
    ctx.clearRect(0, 0, 60, 60);
    
    // Background circle
    ctx.beginPath();
    ctx.arc(centerX, centerY, radius, 0, Math.PI * 2);
    ctx.fillStyle = '#333';
    ctx.fill();
    ctx.strokeStyle = '#555';
    ctx.lineWidth = 2;
    ctx.stroke();
    
    // Value arc (270° range)
    const startAngle = (135 * Math.PI) / 180;
    const angleRange = (270 * Math.PI) / 180;
    const currentAngle = startAngle + (value / 100) * angleRange;
    
    ctx.beginPath();
    ctx.arc(centerX, centerY, radius - 3, startAngle, currentAngle);
    ctx.strokeStyle = '#00d4ff'; // Cyan
    ctx.lineWidth = 4;
    ctx.stroke();
    
    // Center indicator line
    ctx.save();
    ctx.translate(centerX, centerY);
    ctx.rotate(currentAngle);
    ctx.beginPath();
    ctx.moveTo(0, -radius + 8);
    ctx.lineTo(0, -radius + 18);
    ctx.strokeStyle = '#fff';
    ctx.lineWidth = 2;
    ctx.stroke();
    ctx.restore();
}

function setupRotaryKnob(canvas, trackIndex) {
    let isDragging = false;
    let startY, startValue;
    
    canvas.addEventListener('mousedown', (e) => {
        isDragging = true;
        startY = e.clientY;
        startValue = trackVolumes[trackIndex];
    });
    
    document.addEventListener('mousemove', (e) => {
        if (!isDragging) return;
        
        const deltaY = startY - e.clientY; // Inverted
        const newValue = Math.max(0, Math.min(100, startValue + deltaY * 0.5));
        
        trackVolumes[trackIndex] = newValue;
        drawRotaryKnob(canvas, newValue);
        updateVolumeLabel(trackIndex, newValue);
    });
    
    document.addEventListener('mouseup', () => {
        isDragging = false;
    });
}
```

### Audio Playback
```javascript
// Planned V3.1.0
async function launchCell(track, scene) {
    const assignment = cellAssignments[`${track}-${scene}`];
    if (!assignment) return;
    
    // Load audio file from LIBRARY
    const audioFile = await getAudioFromLibrary(assignment.audioFilename);
    const buffer = await audioContext.decodeAudioData(await audioFile.arrayBuffer());
    
    // Create source for this track
    const source = audioContext.createBufferSource();
    source.buffer = buffer;
    source.connect(trackGains[track]);
    source.start();
    
    // Store active playback
    activePlayback[track] = {
        source: source,
        scene: scene,
        startTime: audioContext.currentTime
    };
    
    // Start shader rendering for this track
    startShaderRendering(track, assignment.shaderCode);
}
```

### Shader Rendering
```javascript
// Planned V3.1.0
function startShaderRendering(track, shaderCode) {
    // Create WebGPU pipeline for this track
    const pipeline = createPipelineForTrack(track, shaderCode);
    
    // Render to dedicated canvas
    const canvas = trackCanvases[track];
    const context = canvas.getContext('webgpu');
    
    // Animation loop
    function render(time) {
        // Update uniforms with track audio data
        const fftData = getTrackFFTData(track);
        updateShaderUniforms(pipeline, time, fftData);
        
        // Render frame
        renderShaderFrame(pipeline, context);
        
        // Continue if track still playing
        if (activePlayback[track]) {
            requestAnimationFrame(render);
        }
    }
    
    requestAnimationFrame(render);
}
```
